 IS

CREATE OR REPLACE FUNCTION count_words(sentence VARCHAR2)
RETURN NUMBER
IS
    word_count NUMBER := 0;
BEGIN
    IF sentence IS NULL OR TRIM(sentence) = '' THEN
        RETURN 0;
    END IF;

    -- Count spaces
    FOR i IN 1..LENGTH(sentence) LOOP
        IF SUBSTR(sentence, i, 1) = ' ' THEN
            word_count := word_count + 1;
        END IF;
    END LOOP;

    RETURN word_count + 1;
END;
/
